### Title
Unauthenticated LOOP plugin pprof/metrics/discovery endpoints expose internal debug data without authentication - (File: core/web/router.go, core/web/loop_registry.go)

### Summary
The CVE describes InvoicePlane exposing files/directories to unauthenticated users due to a misconfigured web server lacking auth on sensitive paths. The closest analog in this chainlink repo is the `loopRoutes` registration, which mounts `/discovery`, `/plugins/:name/metrics`, `/plugins/:name/debug/pprof/*profile`, and `/plugins/:name/debug/pprof/symbol` onto the `api` router group without any `auth.Authenticate(...)` middleware, unlike other route groups such as `v2Routes` or `sessionRoutes`.

### Finding Description
`NewRouter` builds an `api` group with only rate limiting and session-cookie middleware attached [1](#0-0) . Compare this to `sessionRoutes`, which explicitly wraps its authenticated routes with `auth.Authenticate(app.AuthenticationProvider(), auth.AuthenticateBySession)` before allowing access [2](#0-1) . `loopRoutes`, however, registers its handlers directly on the bare `api` group with no equivalent authentication wrapper [3](#0-2) .

These handlers — `discoveryHandler`, `pluginMetricHandler`, `pluginPPROFHandler`, and `pluginPPROFPOSTSymbolHandler` — proxy requests to internal LOOP plugin ports and directly write the proxied response bytes back to the client [4](#0-3) [5](#0-4) . The pprof handler builds the backend URL by directly concatenating the caller-supplied `:profile` path parameter [6](#0-5) , and the comment "unlike discovery, this endpoint is internal btw the node and plugin" acknowledges these were designed assuming internal-only reachability, not that they are protected by authentication.

### Impact Explanation
An unauthenticated client hitting these public HTTP endpoints can enumerate registered LOOP plugins, retrieve internal Prometheus `/metrics` output, and trigger/fetch `pprof` debug profiles (heap, goroutine, cpu profile, symbol resolution) from arbitrary running plugin processes — all without any session cookie or API key. This is directly analogous to the CVE's "unauthenticated directory listing and file download," in that internal, non-public operational/debug data intended to be inaccessible externally is exposed to any unauthenticated caller who can reach the node's web server.

### Likelihood Explanation
Likelihood is high if these routes are reachable on the node's externally-facing web server port, since no authentication check gates them and no code elsewhere in `router.go` appears to add one before `loopRoutes` executes.

### Recommendation
Wrap `loopRoutes(app, api)` registration with the same `auth.Authenticate(...)` middleware used for other privileged routes (e.g., `v2Routes`), or move these debug/metrics endpoints to an internal-only listener/port that is not exposed on the public HTTP interface.

### Proof of Concept
Without valid session/API credentials, send:
```
GET /discovery
GET /plugins/<name>/metrics
GET /plugins/<name>/debug/pprof/heap
```
to the node's web server. Because `loopRoutes` is mounted on `api` with no `auth.Authenticate` call, these requests are expected to succeed and return plugin metrics/pprof data despite the absence of credentials — this could not be fully confirmed by tracing every middleware layering due to indexing limits, so exact behavior should be validated by running the node and issuing the above requests directly.

func sessionRoutes(app chainlink.Application, r *gin.RouterGroup) {
	config := app.GetConfig()
	rl := config.WebServer().RateLimit()
	unauth := r.Group("/", rateLimiter(
		rl.UnauthenticatedPeriod(),
		rl.Unauthenticated(),
	))
	sc := NewSessionsController(app)
	unauth.POST("/sessions", sc.Create)
	auth := r.Group("/", auth.Authenticate(app.AuthenticationProvider(), auth.AuthenticateBySession))
	auth.DELETE("/sessions", sc.Destroy)
}

func loopRoutes(app chainlink.Application, r *gin.RouterGroup) {
	loopRegistry := NewLoopRegistryServer(app)
	r.GET("/discovery", ginHandlerFromHTTP(loopRegistry.discoveryHandler))
	r.GET("/plugins/:name/metrics", loopRegistry.pluginMetricHandler)
	r.GET("/plugins/:name/debug/pprof/*profile", loopRegistry.pluginPPROFHandler)
	r.POST("/plugins/:name/debug/pprof/symbol", loopRegistry.pluginPPROFPOSTSymbolHandler)
}

func (l *LoopRegistryServer) pluginMetricHandler(gc *gin.Context) {
	pluginName := gc.Param("name")
	p, ok := l.registry.Get(pluginName)
	if !ok {
		gc.Data(http.StatusNotFound, "text/plain", fmt.Appendf(nil, "plugin %q does not exist", html.EscapeString(pluginName)))
		return
	}

	// unlike discovery, this endpoint is internal btw the node and plugin
	pluginURL := fmt.Sprintf("http://%s:%d/metrics", l.loopHostName, p.EnvCfg.PrometheusPort)
	req, err := http.NewRequestWithContext(gc.Request.Context(), http.MethodGet, pluginURL, nil)
	if err != nil {
		gc.Data(http.StatusInternalServerError, "text/plain", fmt.Appendf(nil, "error creating plugin metrics request: %s", err))
		return
	}
	res, err := l.promClient.Do(req)
	if err != nil {
		msg := "plugin metric handler failed to get plugin url " + html.EscapeString(pluginURL)
		l.logger.Errorw(msg, "err", err)
		gc.Data(http.StatusInternalServerError, "text/plain", fmt.Appendf(nil, "%s: %s", msg, err))
		return
	}
	defer res.Body.Close()
	b, err := io.ReadAll(res.Body)
	if err != nil {
		msg := fmt.Sprintf("error reading plugin %q metrics", html.EscapeString(pluginName))
		l.logger.Errorw(msg, "err", err)
		gc.Data(http.StatusInternalServerError, "text/plain", fmt.Appendf(nil, "%s: %s", msg, err))
		return
	}

	gc.Data(http.StatusOK, "text/plain", b)
}

func (l *LoopRegistryServer) pluginPPROFHandler(gc *gin.Context) {
	pluginName := gc.Param("name")
	p, ok := l.registry.Get(pluginName)
	if !ok {
		gc.Data(http.StatusNotFound, "text/plain", fmt.Appendf(nil, "plugin %q does not exist", html.EscapeString(pluginName)))
		return
	}

	// unlike discovery, this endpoint is internal btw the node and plugin
	pluginURL := fmt.Sprintf("http://%s:%d/debug/pprof/"+gc.Param("profile"), l.loopHostName, p.EnvCfg.PrometheusPort)
	urlVals, timeout := pprofURLVals(gc)
	if s := urlVals.Encode(); s != "" {
		pluginURL += "?" + s
	}
	l.logger.Infow("Forwarding plugin pprof request", "plugin", pluginName, "url", pluginURL)
	l.doRequest(gc, "GET", pluginURL, nil, timeout, pluginName)
}

func (l *LoopRegistryServer) pluginPPROFPOSTSymbolHandler(gc *gin.Context) {
	pluginName := gc.Param("name")
	p, ok := l.registry.Get(pluginName)
	if !ok {
		gc.Data(http.StatusNotFound, "text/plain", fmt.Appendf(nil, "plugin %q does not exist", html.EscapeString(pluginName)))
		return
	}

	// unlike discovery, this endpoint is internal btw the node and plugin
	pluginURL := fmt.Sprintf("http://%s:%d/debug/pprof/symbol", l.loopHostName, p.EnvCfg.PrometheusPort)
	urlVals, timeout := pprofURLVals(gc)
	if s := urlVals.Encode(); s != "" {
		pluginURL += "?" + s
	}
	body, err := io.ReadAll(gc.Request.Body)
	if err != nil {
		gc.Data(http.StatusInternalServerError, "text/plain", fmt.Appendf(nil, "error reading plugin pprof request body: %s", err))
		return
	}
	l.doRequest(gc, "POST", pluginURL, bytes.NewReader(body), timeout, pluginName)
}

func (l *LoopRegistryServer) doRequest(gc *gin.Context, method, url string, body io.Reader, timeout time.Duration, pluginName string) {
	ctx, cancel := context.WithTimeout(gc.Request.Context(), timeout)
	defer cancel()
	req, err := http.NewRequestWithContext(ctx, method, url, body)
	if err != nil {
		gc.Data(http.StatusInternalServerError, "text/plain", fmt.Appendf(nil, "error creating plugin pprof request: %s", err))
		return
	}
	res, err := http.DefaultClient.Do(req)
	if err != nil {
		msg := "plugin pprof handler failed to post plugin url " + html.EscapeString(url)
		l.logger.Errorw(msg, "err", err)
		gc.Data(http.StatusInternalServerError, "text/plain", fmt.Appendf(nil, "%s: %s", msg, err))
		return
	}
	defer res.Body.Close()
	b, err := io.ReadAll(res.Body)
	if err != nil {
		msg := fmt.Sprintf("error reading plugin %q pprof", html.EscapeString(pluginName))
		l.logger.Errorw(msg, "err", err)
		gc.Data(http.StatusInternalServerError, "text/plain", fmt.Appendf(nil, "%s: %s", msg, err))
		return
	}

	gc.Data(http.StatusOK, "text/plain", b)
}
