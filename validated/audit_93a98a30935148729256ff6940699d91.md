Based on the investigation, this CVE describes a kubevirt-specific privilege escalation where a VM process gains host filesystem access — a virtualization/sandbox-escape bug class with no meaningful analog in alt.fun's Solidity contracts.

Checking the actual access-control surfaces in scope:
- `Router` gates all state-mutating entry points (`buy`, `sell`, `addInitialLiquidity`, `graduate`) behind `onlyRole(BONDING_ROLE)`, held only by `Bonding`. [1](#0-0) [2](#0-1) 
- `Pair` restricts `mint`, `swap`, `transferAsset`, `transferToken` to `onlyRouter`, and `router` is immutable, set once from `Factory.router` at construction. [3](#0-2) 
- `FeeVault.accrue` is gated by `onlyDepositor`, an owner-controlled allowlist; claim paths (`claim`, `claimProtocol`, `sweepDonations`) always pay to the caller's own accrued balance or the fixed `feeTo` address, never an attacker-chosen recipient.

    modifier onlyRouter() {
        if (msg.sender != router) revert OnlyRouter();
        _;
    }

    constructor(
        address router_,
        address launchedToken_,
        address assetToken_
    ) {
        if (router_ == address(0) || launchedToken_ == address(0) || assetToken_ == address(0)) revert ZeroAddress();
        if (launchedToken_ == assetToken_) revert IdenticalTokens();
        router = router_;
        launchedToken = launchedToken_;
        assetToken = assetToken_;
    }

    function mint(
        uint256 tokenReserve,
        uint256 assetReserve
    ) external onlyRouter returns (bool) {
        if (_pool.k != 0) revert AlreadyMinted();
        _pool = Pool({tokenReserve: tokenReserve, assetReserve: assetReserve, k: tokenReserve * assetReserve});
        emit Mint(tokenReserve, assetReserve);
        return true;
    }

    function swap(
        uint256 tokenIn,
        uint256 tokenOut,
        uint256 assetIn,
        uint256 assetOut
    ) external onlyRouter returns (bool) {
        uint256 newTokenReserve = (_pool.tokenReserve + tokenIn) - tokenOut;
        uint256 newAssetReserve = (_pool.assetReserve + assetIn) - assetOut;
        if ((newTokenReserve + 1) * (newAssetReserve + 1) < _pool.k) revert KInvariantViolated();

        _pool.tokenReserve = newTokenReserve;
        _pool.assetReserve = newAssetReserve;
        emit Swap(tokenIn, tokenOut, assetIn, assetOut);
        return true;
    }

    function transferAsset(
        address recipient,
        uint256 amount
    ) external onlyRouter {
        IERC20(assetToken).safeTransfer(recipient, amount);
    }
