### Analysis

`rest.WrapUpdatedObjectInfo` (used by `Storage.Update`) composes update pipelines by first calling the *inner* `UpdatedObjectInfo.UpdatedObject()` to completion, and only then invoking its own additional transformers on the *result*: [1](#0-0) 

For a Server-Side-Apply PATCH, the request handler builds the update pipeline as `p.updatedObjectInfo = rest.DefaultUpdatedObjectInfo(nil, p.applyPatch, p.applyAdmission, dedupOwnerReferencesTransformer)` and passes that into `p.restPatcher.Update(...)`, where `p.restPatcher` is the RBAC `policybased.Storage`: [2](#0-1) 

`p.applyPatch` is exactly the transform that performs the full SSA structured-merge-diff merge against the currently persisted object via the field manager: [3](#0-2) 

`Storage.Update` wraps that pipeline with `rest.WrapUpdatedObjectInfo(obj, nonEscalatingTransform)`: [4](#0-3) 

Because `wrappedUpdatedObjectInfo.UpdatedObject` calls `i.objInfo.UpdatedObject(ctx, oldObj)` **first** — which runs `applyPatch` → `applyAdmission` → dedup, producing the fully-merged final object — and only *afterward* runs the escalation-check transform on that already-merged result, the `obj.(*rbac.ClusterRoleBinding)` seen by `ConfirmNoEscalation`/`BindingAuthorized` is the final, post-merge `roleRef`, not a pre-merge or partial view. There is no separate "diff-based" view that could diverge from the persisted merge result; the check operates on the same object that `GuaranteedUpdate` will attempt to persist: [5](#0-4) 

The premise that SSA merging happens "after admission" and separately from what the escalation guard inspects is incorrect for this code path — the merge (`applyPatch`) executes as the first transformer in the chain that `Storage.Update`'s wrapper delegates to before applying its own escalation check, so the guard always sees the fully merged `roleRef`.

func (i *wrappedUpdatedObjectInfo) UpdatedObject(ctx context.Context, oldObj runtime.Object) (runtime.Object, error) {
	newObj, err := i.objInfo.UpdatedObject(ctx, oldObj)
	if err != nil {
		return newObj, err
	}

	// Allow any configured transformers to update the new object or error
	for _, transformer := range i.transformers {
		newObj, err = transformer(ctx, newObj, oldObj)
		if err != nil {
			return nil, err
		}
	}

func (s *Storage) Update(ctx context.Context, name string, obj rest.UpdatedObjectInfo, createValidation rest.ValidateObjectFunc, updateValidation rest.ValidateObjectUpdateFunc, forceAllowCreate bool, options *metav1.UpdateOptions) (runtime.Object, bool, error) {
	if rbacregistry.EscalationAllowed(ctx) {
		return s.StandardStorage.Update(ctx, name, obj, createValidation, updateValidation, forceAllowCreate, options)
	}

	nonEscalatingInfo := rest.WrapUpdatedObjectInfo(obj, func(ctx context.Context, obj runtime.Object, oldObj runtime.Object) (runtime.Object, error) {
		clusterRoleBinding := obj.(*rbac.ClusterRoleBinding)

		// if we're only mutating fields needed for the GC to eventually delete this obj, return
		if rbacregistry.IsOnlyMutatingGCFields(obj, oldObj, kapihelper.Semantic) {
			return obj, nil
		}

		// if we're explicitly authorized to bind this clusterrole, return
		if rbacregistry.BindingAuthorized(ctx, clusterRoleBinding.RoleRef, metav1.NamespaceNone, s.authorizer) {
			return obj, nil
		}

		// Otherwise, see if we already have all the permissions contained in the referenced clusterrole
		v1RoleRef := rbacv1.RoleRef{}
		err := rbacv1helpers.Convert_rbac_RoleRef_To_v1_RoleRef(&clusterRoleBinding.RoleRef, &v1RoleRef, nil)
		if err != nil {
			return nil, err
		}
		rules, err := s.ruleResolver.GetRoleReferenceRules(ctx, v1RoleRef, metav1.NamespaceNone)
		if err != nil {
			return nil, err
		}
		if err := rbacregistryvalidation.ConfirmNoEscalation(ctx, s.ruleResolver, rules); err != nil {
			return nil, errors.NewForbidden(groupResource, clusterRoleBinding.Name, err)
		}
		return obj, nil
	})

	return s.StandardStorage.Update(ctx, name, nonEscalatingInfo, createValidation, updateValidation, forceAllowCreate, options)
