Based on my research, alt.fun's contracts have no analog to the balance-based, per-address unbonding timer that the Hubble `unbondShares()` bug exploits. The relevant candidates don't share the vulnerable pattern:

- `LPLock.recordLock` is a one-shot, locker-gated lock with **no withdraw path at all** ("No withdraw in v1"), so there's no time-gated withdrawal that a share transfer could bypass. [1](#0-0) [2](#0-1) 

- `FeeVault.claim()` and `claimProtocol()` pay out immediately against a `creatorBalance`/`protocolBalance` mapping keyed by address with no time delay or unbonding period to bypass. [3](#0-2) 

- `Bonding.transferCreator` reassigns the creator role instantly but doesn't gate any withdrawal behind a timer, and `FeeVault` balances stay keyed to whichever address earned them rather than following a transferable "share" balance. [4](#0-3) 

None of the in-scope reachable functions (`Zap.buy/sell`, `Bonding.triggerGraduation/finalizeGraduation/transferCreator`, `FeeVault.claim/claimProtocol/sweepDonations`, or direct token/LT transfers) implement a "mint shares → start unbonding timer → check balanceOf(msg.sender) → transfer shares to reset timer" pattern. There is no unbonding mechanism in alt.fun's codebase for this bug class to attach to.

```text
    function recordLock(
        address token,
        address lpPair,
        uint256 amount
    ) external {
        LPLockStorage storage $ = _s();
        if (!$.isLocker[msg.sender]) revert NotAuthorized();
        if (lpPair == address(0)) revert ZeroAddress();
        if (amount == 0) revert ZeroAmount();
        // `lockedAt` is the one-shot sentinel: it is always set to a non-zero
        // timestamp on the first lock, so the guard holds for any `amount`.
        if ($.locks[token].lockedAt != 0) revert AlreadyLocked();
        if (IERC20(lpPair).balanceOf(address(this)) < amount) revert InsufficientLPBalance();
        $.locks[token] = LockInfo({lpPair: lpPair, amount: amount, lockedAt: block.timestamp});
        emit LPLocked(token, lpPair, amount);
    }
```

```text
    function claim() external nonReentrant returns (uint256 amount) {
        FeeVaultStorage storage $ = _s();
        amount = $.creatorBalance[msg.sender];
        if (amount == 0) revert NothingToClaim();
        $.creatorBalance[msg.sender] = 0;
        $.totalAccruedCreator -= amount;
        $.usdc.safeTransfer(msg.sender, amount);
        emit CreatorFeesClaimed(msg.sender, amount);
    }

    function claimProtocol() external nonReentrant returns (uint256 amount) {
        FeeVaultStorage storage $ = _s();
        amount = $.protocolBalance;
        if (amount == 0) revert NothingToClaim();
        $.protocolBalance = 0;
        address feeTo_ = $.feeTo;
        $.usdc.safeTransfer(feeTo_, amount);
        emit ProtocolFeesClaimed(feeTo_, amount);
    }
```

```text
    function transferCreator(
        address tokenAddress,
        address newCreator
    ) external {
        if (newCreator == address(0)) revert ZeroAddress();
        TokenInfo storage info = _s().tokenInfo[tokenAddress];
        if (msg.sender != info.creator) revert NotCreator();
        if (newCreator == info.creator) revert InvalidInput();
        info.creator = newCreator;
        emit CreatorTransferred(tokenAddress, msg.sender, newCreator);
    }
```
