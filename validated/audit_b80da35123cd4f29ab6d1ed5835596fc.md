### No Vulnerability found for this question.

**Rationale:** The hardcoded `attrs.APIVersion = "*"` in `BindingAuthorized` (and `RoleEscalationAuthorized`) at `pkg/registry/rbac/escalation_check.go` lines 83, 126, and 131 is not exploitable because the built-in RBAC authorizer never inspects `APIVersion` when evaluating rules. `RuleAllows` in [1](#0-0)  only matches on verb, API group, resource, and resource name — via `VerbMatches`, `APIGroupMatches`, `ResourceMatches`, and `ResourceNameMatches` from [2](#0-1)  the `rbacv1helpers` package (see [3](#0-2) ). There is no `APIVersion`-based check anywhere in the in-tree RBAC evaluation path, so the `"*"` value in `attrs.APIVersion` has zero effect on RBAC decisions — it's neither read nor compared.

The security boundary in Kubernetes RBAC for "bind"/"escalate" checks is intentionally `Group + Resource` (not `Group + Version + Resource`), because API versions are just different serializations of the same underlying resource/identity, not separate authorization scopes. This is consistent with how RBAC `PolicyRule.APIGroups`/`Resources` are defined (no `APIVersions` field exists in `rbacv1.PolicyRule` at all) — see [4](#0-3) . Since RBAC rules cannot be scoped to a specific `apiVersion` in the first place, the premise of "a webhook or RBAC rule that scopes 'bind' to a specific apiVersion string" doesn't apply to the standard RBAC authorizer path that `BindingAuthorized` exercises.

While a hypothetical *external webhook authorizer* could choose to inspect the `apiVersion` field from a `SubjectAccessReview` and apply version-specific policy, that is a webhook-implementation/operator-configuration choice outside the scope of this codebase and outside the unprivileged-attacker threat model (no exploitable code path is reachable purely from an ordinary authenticated user's bind/escalate request through this function without a specially-configured, non-default third-party authorizer). This matches upstream Kubernetes' original design (`k8s.io/kubernetes`) and is not a defect introduced or exploitable within this repository's control-plane code.

func RuleAllows(requestAttributes authorizer.Attributes, rule *rbacv1.PolicyRule) bool {
	if requestAttributes.IsResourceRequest() {
		combinedResource := requestAttributes.GetResource()
		if len(requestAttributes.GetSubresource()) > 0 {
			combinedResource = requestAttributes.GetResource() + "/" + requestAttributes.GetSubresource()
		}

		return rbacv1helpers.VerbMatches(rule, requestAttributes.GetVerb()) &&
			rbacv1helpers.APIGroupMatches(rule, requestAttributes.GetAPIGroup()) &&
			rbacv1helpers.ResourceMatches(rule, combinedResource, requestAttributes.GetSubresource()) &&
			rbacv1helpers.ResourceNameMatches(rule, requestAttributes.GetName())
	}

	return rbacv1helpers.VerbMatches(rule, requestAttributes.GetVerb()) &&
		rbacv1helpers.NonResourceURLMatches(rule, requestAttributes.GetPath())

}

func APIGroupMatches(rule *rbacv1.PolicyRule, requestedGroup string) bool {
	for _, ruleGroup := range rule.APIGroups {
		if ruleGroup == rbacv1.APIGroupAll {
			return true
		}
		if ruleGroup == requestedGroup {
			return true
		}
	}

	return false
}

func ResourceMatches(rule *rbacv1.PolicyRule, combinedRequestedResource, requestedSubresource string) bool {
	for _, ruleResource := range rule.Resources {
		// if everything is allowed, we match
		if ruleResource == rbacv1.ResourceAll {
			return true
		}
		// if we have an exact match, we match
		if ruleResource == combinedRequestedResource {
			return true
		}

		// We can also match a */subresource.
		// if there isn't a subresource, then continue
		if len(requestedSubresource) == 0 {
			continue
		}
		// if the rule isn't in the format */subresource, then we don't match, continue
		if len(ruleResource) == len(requestedSubresource)+2 &&
			strings.HasPrefix(ruleResource, "*/") &&
			strings.HasSuffix(ruleResource, requestedSubresource) {
			return true

		}
	}

	return false
}

func ResourceNameMatches(rule *rbacv1.PolicyRule, requestedName string) bool {
	if len(rule.ResourceNames) == 0 {
		return true
	}

	for _, ruleName := range rule.ResourceNames {
		if ruleName == requestedName {
			return true
		}
	}

	return false
}
