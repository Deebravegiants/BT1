    fn challenge_128bit_scalars(&mut self, label: &[u8], num_scalars: usize) -> Vec<F> {
        let mut buf = vec![0u8; num_scalars * 16];
        self.challenge_bytes(label, &mut buf);

        buf.chunks(16)
            .map(|chunk| F::from_le_bytes_mod_order(chunk.try_into().unwrap()))
            .collect()
    }
