    public fun slice<Element: copy>(
        self: &vector<Element>,
        start: u64,
        end: u64
    ): vector<Element> {
        assert!(start <= end && end <= self.length(), EINVALID_SLICE_RANGE);

        let vec = vector[];
        while (start < end) {
            vec.push_back(self[start]);
            start += 1;
        };
        vec
    }
